cmake_minimum_required(VERSION 3.21)
project(R387Agent VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

option(R387_CORE_TESTS_ONLY "Build and test the portable image-conversion engine without Geode" OFF)
if (R387_CORE_TESTS_ONLY)
    add_executable(R387CoreTests tests/core_tests.cpp src/ImageProcessor.cpp)
    target_include_directories(R387CoreTests PRIVATE src)
    enable_testing()
    add_test(NAME R387CoreTests COMMAND R387CoreTests)
    return()
endif()

set(CMAKE_CXX_VISIBILITY_PRESET hidden)
if (NOT WIN32)
    message(FATAL_ERROR "R387 Agent currently targets Geometry Dash for Windows only. Use -DR387_CORE_TESTS_ONLY=ON to run the portable engine tests.")
endif()
if (NOT DEFINED ENV{GEODE_SDK})
    message(FATAL_ERROR "GEODE_SDK is not set. Install Geode CLI and run 'geode sdk install'.")
endif()

file(GLOB_RECURSE R387_SOURCES CONFIGURE_DEPENDS src/*.cpp)
add_library(R387Agent SHARED ${R387_SOURCES})
target_include_directories(R387Agent PRIVATE src)
target_link_libraries(R387Agent gdiplus comdlg32)
add_subdirectory($ENV{GEODE_SDK} ${CMAKE_CURRENT_BINARY_DIR}/geode)
setup_geode_mod(R387Agent)
